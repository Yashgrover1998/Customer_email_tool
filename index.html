How to Connect Your Gmail Inbox & Clear Bulk Backlogs

Convert thousands of pending unread support emails into structured tickets automatically using Google Sheets as your real-time database and Google Apps Script as your REST API & email dispatcher.

Stage 1 • Ingestion

Gmail Support Inbox

Apps Script scans is:unread threads in batches, extracts sender, subject, body, and Gmail Thread ID, and applies label Ticketed.

Stage 2 • Auto-Triage

Smart Keyword Classifier

Automatically assigns Priority (Urgent, High) and Category (Billing, Technical) based on email keywords.

Stage 3 • Database

Google Sheets DB + API

Stores every ticket in a structured sheet and exposes doGet() and doPost() JSON endpoints for this HTML dashboard.

Stage 4 • Bulk Closure

1-Click Thread Reply

Resolving a ticket (single or bulk) replies directly inside the customer's Gmail thread and updates the Sheet status to Resolved.

Pro Tip for 1,000+ Pending Emails:

Gmail Apps Script limits execution to 6 minutes per run. The script below processes 50 unread threads per batch (MAX_BATCH_SIZE = 50) and runs every 5 minutes—safely converting 600+ emails/hour without hitting Google quota limits.

Code.gs — Google Apps Script Backend 
/**
 * PULSEDESK — GMAIL TO GOOGLE SHEETS TICKET ENGINE & WEB APP API
 * 1. Converts unread Gmail threads into structured tickets in batches
 * 2. Auto-assigns Priority & Category using keyword triage rules
 * 3. Serves JSON API (doGet / doPost) for the HTML Ticket Dashboard
 */

const SHEET_NAME = 'Tickets';
const PROCESSED_LABEL = 'Ticketed';
const MAX_BATCH_SIZE = 50; // Safe batch size for large email backlogs

/**
 * RUN THIS ONCE: Initializes Sheet headers, Gmail label & 5-min cron trigger
 */
function setupSheetAndTrigger() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'TicketID', 'ThreadID', 'Timestamp', 'CustomerName',
      'CustomerEmail', 'Subject', 'Body', 'Category',
      'Priority', 'Status', 'Resolution', 'ResolvedAt'
    ]);
    sheet.getRange('A1:L1').setFontWeight('bold').setBackground('#f1f5f9');
  }
  // Ensure Gmail label exists
  if (!GmailApp.getUserLabelByName(PROCESSED_LABEL)) {
    GmailApp.createLabel(PROCESSED_LABEL);
  }
  // Ingest initial batch right away
  ingestUnreadEmailsToTickets();

  // Create 5-minute automated trigger if not already set
  const triggers = ScriptApp.getProjectTriggers();
  const exists = triggers.some(t => t.getHandlerFunction() === 'ingestUnreadEmailsToTickets');
  if (!exists) {
    ScriptApp.newTrigger('ingestUnreadEmailsToTickets')
      .timeBased()
      .everyMinutes(5)
      .create();
  }
}

/**
 * CORE ENGINE: Pulls unread emails, classifies priority/category & logs tickets
 */
function ingestUnreadEmailsToTickets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  let label = GmailApp.getUserLabelByName(PROCESSED_LABEL);
  if (!label) label = GmailApp.createLabel(PROCESSED_LABEL);

  // Query unread inbox emails that haven't been converted yet
  const query = 'is:unread label:inbox -label:' + PROCESSED_LABEL;
  const threads = GmailApp.search(query, 0, MAX_BATCH_SIZE);
  if (threads.length === 0) return 0;

  const newRows = [];
  threads.forEach(thread => {
    const messages = thread.getMessages();
    const firstMsg = messages[0];
    const rawFrom = firstMsg.getFrom();
    const parsedSender = parseSenderString(rawFrom);
    const subject = firstMsg.getSubject() || '(No Subject)';
    const body = firstMsg.getPlainBody().substring(0, 4000);
    const triage = autoClassifyEmail(subject, body);
    const ticketId = 'TKT-' + Math.floor(1000 + Math.random() * 9000);

    newRows.push([
      ticketId,
      thread.getId(),
      firstMsg.getDate().toISOString(),
      parsedSender.name,
      parsedSender.email,
      subject,
      body,
      triage.category,
      triage.priority,
      'Open',
      '',
      ''
    ]);

    thread.addLabel(label);
    thread.markRead();
  });

  // Bulk write to Google Sheet in a single fast I/O operation
  if (newRows.length > 0) {
    sheet.getRange(sheet.getLastRow() + 1, 1, newRows.length, 12).setValues(newRows);
  }
  return newRows.length;
}

/**
 * KEYWORD TRIAGE CLASSIFIER: Assigns Priority & Category automatically
 */
function autoClassifyEmail(subject, body) {
  const text = (subject + ' ' + body).toLowerCase();
  let priority = 'Medium';
  let category = 'General';

  if (/urgent|outage|down|charged twice|unauthorized|breach|asap|critical/.test(text)) {
    priority = 'Urgent';
  } else if (/error|failed|refund|invoice|locked|cannot login|bug/.test(text)) {
    priority = 'High';
  } else if (/question|how to|feature|feedback/.test(text)) {
    priority = 'Low';
  }

  if (/invoice|refund|charge|billing|payment|subscription/.test(text)) {
    category = 'Billing';
  } else if (/bug|error|crash|api|500|broken|slow/.test(text)) {
    category = 'Technical';
  } else if (/login|password|2fa|account|locked|access/.test(text)) {
    category = 'Account';
  } else if (/order|shipping|tracking|delivery|package/.test(text)) {
    category = 'Shipping';
  }

  return { priority, category };
}

function parseSenderString(raw) {
  const match = raw.match(/(.*)<(.+)>/);
  if (match) {
    return { name: match[1].replace(/"/g, '').trim() || match[2], email: match[2].trim() };
  }
  return { name: raw.split('@')[0], email: raw.trim() };
}

/**
 * GET API: Fetches all tickets for the HTML Dashboard
 */
function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) || 'list';
  if (action === 'sync') {
    const ingested = ingestUnreadEmailsToTickets();
    return jsonResponse({ status: 'ok', ingestedCount: ingested, tickets: getSheetTickets() });
  }
  return jsonResponse({ status: 'ok', tickets: getSheetTickets() });
}

/**
 * POST API: Handles Single Reply & Resolve, Bulk Resolve, or Status Updates
 */
function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME);
    const data = sheet.getDataRange().getValues();

    if (payload.action === 'resolve' || payload.action === 'bulkResolve') {
      const targetIds = payload.ticketIds || [payload.ticketId];
      const replyBody = payload.replyMessage;
      const sendEmail = payload.sendEmail !== false;
      const nowIso = new Date().toISOString();

      for (let i = 1; i < data.length; i++) {
        const rowTicketId = data[i][0];
        if (targetIds.indexOf(rowTicketId) !== -1) {
          const threadId = data[i][1];
          // Reply directly to the customer's Gmail thread
          if (sendEmail && threadId && replyBody) {
            try {
              const thread = GmailApp.getThreadById(threadId);
              if (thread) thread.reply(replyBody);
            } catch (mailErr) {
              // Fallback to sendEmail if thread ID not found
              GmailApp.sendEmail(data[i][4], 'Re: ' + data[i][5] + ' [' + rowTicketId + ']', replyBody);
            }
          }
          sheet.getRange(i + 1, 10).setValue('Resolved');
          sheet.getRange(i + 1, 11).setValue(replyBody);
          sheet.getRange(i + 1, 12).setValue(nowIso);
        }
      }
      return jsonResponse({ status: 'ok', updated: targetIds.length });
    }
    return jsonResponse({ status: 'ok' });
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

function getSheetTickets() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sheet || sheet.getLastRow() <= 1) return [];
  const rows = sheet.getDataRange().getValues();
  rows.shift();
  return rows.map(r => ({
    id: r[0], threadId: r[1], timestamp: r[2], customerName: r[3],
    customerEmail: r[4], subject: r[5], body: r[6], category: r[7],
    priority: r[8], status: r[9], resolution: r[10], resolvedAt: r[11]
  })).reverse();
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
